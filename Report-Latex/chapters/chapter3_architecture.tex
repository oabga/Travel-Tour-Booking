\section{Tổng quan kiến trúc}\label{Sec3.1}
% ═════════════════════════════════════════════════════════════

\begin{de}\label{def3.1}
Kiến trúc 3 tầng (3-Layer Architecture) là mô hình tổ chức phần
mềm phân tách ứng dụng thành ba lớp độc lập, mỗi lớp chỉ giao
tiếp với lớp kề nó thông qua Interface đã định nghĩa trước.
Nguyên tắc này đảm bảo tính \textit{loose coupling} (kết nối
lỏng lẻo) và \textit{high cohesion} (gắn kết cao trong nội bộ),
giúp hệ thống dễ bảo trì, kiểm thử và mở rộng độc lập từng tầng.
\end{de}

Hệ thống Travel Tour Booking được tổ chức thành bốn project C\#
trong cùng một Solution, tương ứng với từng tầng kiến trúc và
phần dùng chung, như được mô tả trong Bảng~\ref{table:projects}.

\begin{table}[H]
  \begin{center}
    \begin{tabular}{|l|l|l|}
      \hline
      \textbf{Project} & \textbf{Tầng} & \textbf{Trách nhiệm chính} \\
      \hline
      \texttt{TravelTourBooking.API}
        & Presentation
        & Controllers, Middleware, Swagger, JWT \\
      \hline
      \texttt{TravelTourBooking.BLL}
        & Business Logic
        & Services, Interfaces, Validators, Helpers \\
      \hline
      \texttt{TravelTourBooking.DAL}
        & Data Access
        & DbContext, Entities, Repositories, ADO.NET \\
      \hline
      \texttt{TravelTourBooking.Common}
        & Shared
        & DTOs, ApiResponse, Enums dùng chung \\
      \hline
    \end{tabular}
  \end{center}
  \caption{Bốn project trong Solution và trách nhiệm}
  \label{table:projects}
\end{table}

\begin{thm}\label{thm3.1}
Luồng xử lý trong kiến trúc 3 tầng luôn đi một chiều từ trên
xuống: \texttt{API} $\rightarrow$ \texttt{BLL} $\rightarrow$
\texttt{DAL} $\rightarrow$ \texttt{SQL Server}. Tầng trên không
được phép tham chiếu trực tiếp đến tầng dưới cách một tầng.
\end{thm}

\noindent\begin{proof}
Quy tắc này được thực thi qua cấu hình phụ thuộc project
(\texttt{.csproj}): \texttt{API} chỉ tham chiếu \texttt{BLL}
và \texttt{Common}; \texttt{BLL} chỉ tham chiếu \texttt{DAL}
và \texttt{Common}; \texttt{DAL} chỉ tham chiếu \texttt{Common}.
Do đó \texttt{API} không thể gọi trực tiếp \texttt{AppDbContext}
hay bất kỳ Repository nào — mọi thao tác phải đi qua Interface
của BLL. \QEDAa
\end{proof}

\section{Lớp Giao diện — Presentation Layer}\label{Sec3.2}
% ═════════════════════════════════════════════════════════════

Lớp Giao diện (Presentation Layer) là điểm tiếp nhận mọi
yêu cầu HTTP từ client. Lớp này không chứa bất kỳ logic
nghiệp vụ nào — nhiệm vụ duy nhất là nhận request, gọi
xuống BLL và trả về response chuẩn hóa cho client.

% ──────────────────────────────────────
\subsection{ASP.NET Core Web API và Controllers}\label{sub3.2.1}

\begin{de}\label{def3.2}
ASP.NET Core Web API là framework xây dựng RESTful API trên
nền tảng .NET. Controller là class chứa các action method xử
lý từng HTTP request tương ứng với endpoint cụ thể, được
đánh dấu bằng các attribute như \texttt{[HttpGet]},
\texttt{[HttpPost]}, \texttt{[Authorize]}.
\end{de}

Đoạn~\ref{lst:controller} trình bày \texttt{ToursController}
— Controller quản lý tour với đầy đủ các action CRUD.
Mỗi action nhận DTO từ client, gọi \texttt{ITourService}
và trả về \texttt{ApiResponse<T>} chuẩn hóa.

\begin{lstlisting}[style=csharp,
  caption={ToursController -- CRUD tour với phân quyền},
  label={lst:controller}]
[ApiController]
[Route("api/tours")]
[Produces("application/json")]
public class ToursController : ControllerBase
{
    private readonly ITourService _svc;
    public ToursController(ITourService svc) => _svc = svc;

    // GET api/tours -- phân quyền Admin, Staff
    [HttpGet]
    [Authorize(Roles = "Admin,Staff")]
    [ProducesResponseType(
        typeof(ApiResponse<PagedResult<TourListDto>>), 200)]
    public async Task<IActionResult> GetAll(
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 10,
        [FromQuery] int?     cateId   = null,
        [FromQuery] int?     desId    = null,
        [FromQuery] decimal? priceMin = null,
        [FromQuery] decimal? priceMax = null)
    {
        if (page < 1 || pageSize < 1 || pageSize > 100)
            return BadRequest(ApiResponse<string>
                .Fail("page/pageSize khong hop le."));

        var result = await _svc.GetToursAsync(
            page, pageSize, cateId, desId, priceMin, priceMax);
        return Ok(ApiResponse<PagedResult<TourListDto>>.Ok(result));
    }

    // GET api/tours/search -- công khai, không cần đăng nhập
    [HttpGet("search")]
    [AllowAnonymous]
    public async Task<IActionResult> Search(
        [FromQuery] string?  destination = null,
        [FromQuery] decimal? priceMin    = null,
        [FromQuery] decimal? priceMax    = null,
        [FromQuery] DateOnly? date       = null)
    {
        var results = await _svc.SearchToursAsync(
            destination, priceMin, priceMax, date);
        return Ok(ApiResponse<IEnumerable<object>>
            .Ok(results.Cast<object>()));
    }

    // GET api/tours/{id} -- công khai
    [HttpGet("{id:int}")]
    [AllowAnonymous]
    public async Task<IActionResult> GetById(int id)
    {
        var tour = await _svc.GetTourDetailAsync(id);
        if (tour is null)
            return NotFound(ApiResponse<string>
                .Fail($"Tour ID {id} khong ton tai."));
        return Ok(ApiResponse<TourDetailDto>.Ok(tour));
    }

    // POST api/tours -- chỉ Admin
    [HttpPost]
    [Authorize(Roles = "Admin")]
    public async Task<IActionResult> Create(
        [FromBody] TourRequestDto dto)
    {
        if (!ModelState.IsValid)
            return BadRequest(ApiResponse<string>
                .Fail("Dư lieu khong hop le."));
        var created = await _svc.CreateTourAsync(dto);
        return CreatedAtAction(nameof(GetById),
            new { id = created.TourId },
            ApiResponse<TourDetailDto>.Ok(created,
                "Tao tour thanh cong."));
    }

    // DELETE api/tours/{id} -- chỉ Admin, soft-delete
    [HttpDelete("{id:int}")]
    [Authorize(Roles = "Admin")]
    public async Task<IActionResult> Delete(int id)
    {
        await _svc.DeleteTourAsync(id);
        return Ok(ApiResponse<string>.Ok("deleted",
            "Xoa tour thanh cong."));
    }
}
\end{lstlisting}

\subsection{Middleware Pipeline}\label{sub3.2.2}

\begin{de}\label{def3.3}
Middleware là các thành phần xử lý request và response trong
pipeline của ASP.NET Core. Mỗi middleware có thể xử lý request
trước khi chuyển tiếp cho middleware tiếp theo, hoặc trả về
response ngay lập tức mà không cần đi tiếp.
\end{de}

\texttt{ExceptionMiddleware} trong Đoạn~\ref{lst:middleware}
bắt toàn bộ exception chưa được xử lý trong pipeline, tự động
ánh xạ sang HTTP status code phù hợp và trả về
\texttt{ApiResponse<T>} chuẩn.

\begin{lstlisting}[style=csharp,
  caption={ExceptionMiddleware -- xử lý lỗi tập trung},
  label={lst:middleware}]
public class ExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<ExceptionMiddleware> _logger;

    public ExceptionMiddleware(
        RequestDelegate next,
        ILogger<ExceptionMiddleware> logger)
    { _next = next; _logger = logger; }

    public async Task InvokeAsync(HttpContext ctx)
    {
        try { await _next(ctx); }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "Unhandled exception: {Message}", ex.Message);
            await HandleExceptionAsync(ctx, ex);
        }
    }

    private static Task HandleExceptionAsync(
        HttpContext ctx, Exception ex)
    {
        ctx.Response.ContentType = "application/json";

        // Ánh xạ loại Exception sang HTTP Status Code
        ctx.Response.StatusCode = ex switch
        {
            KeyNotFoundException        => 404, // Not Found
            UnauthorizedAccessException => 403, // Forbidden
            InvalidOperationException   => 409, // Conflict
            ArgumentException           => 400, // Bad Request
            _                           => 500  // Internal Error
        };

        var body = JsonSerializer.Serialize(
            ApiResponse<string>.Fail(
                ex.InnerException?.Message ?? ex.Message),
            new JsonSerializerOptions
            {
                PropertyNamingPolicy = JsonNamingPolicy.CamelCase
            });
        return ctx.Response.WriteAsync(body);
    }
}
\end{lstlisting}

Thứ tự đăng ký middleware trong \texttt{Program.cs} có ý nghĩa
quyết định đến hành vi của hệ thống, như được mô tả trong
Đoạn~\ref{lst:pipeline}.

\begin{lstlisting}[style=csharp,
  caption={Middleware Pipeline trong Program.cs},
  label={lst:pipeline}]
// ExceptionMiddleware dat dau tien de bat moi exception
app.UseMiddleware<ExceptionMiddleware>();

app.UseSwagger();
app.UseSwaggerUI(c =>
{
    c.SwaggerEndpoint("/swagger/v1/swagger.json",
        "TravelTourBooking API v1");
    c.RoutePrefix = "swagger";
});

app.UseHttpsRedirection();
app.UseCors("AllowAngular");    // CORS trước Authentication
app.UseAuthentication();         // Xác thực JWT token
app.UseAuthorization();          // Phân quyền Role
app.MapControllers();
\end{lstlisting}

\begin{cor}
\texttt{UseAuthentication()} phải đứng trước
\texttt{UseAuthorization()} vì phân quyền chỉ thực hiện được
sau khi xác thực danh tính. \texttt{ExceptionMiddleware} đặt
đầu tiên để bắt được mọi exception từ bất kỳ tầng nào trong
pipeline phía sau, kể cả lỗi Authentication.
\end{cor}

\subsection{JWT Authentication và RBAC}\label{sub3.2.3}

\begin{thm}\label{thm3.2}
JWT (JSON Web Token) cho phép xác thực stateless: server không
lưu session, mọi thông tin cần thiết (AccountId, Email, Role)
được mã hóa trong token và gửi kèm mỗi request qua header
\texttt{Authorization: Bearer <token>}.
\end{thm}

\noindent\begin{proof}
Tính stateless được đảm bảo vì server chỉ cần verify chữ ký
(signature) của token bằng \texttt{SecretKey} mà không cần
tra cứu database — giúp hệ thống có thể scale ngang dễ dàng.
\QEDAa
\end{proof}

Khi khách hàng đăng nhập thành công, \texttt{AuthService}
(Đoạn~\ref{lst:auth}) xác minh mật khẩu bằng BCrypt, truy
vấn danh sách Role từ bảng \texttt{AccountRoles} và tạo JWT
token chứa các Claims tương ứng.

\begin{lstlisting}[style=csharp,
  caption={AuthService.LoginAsync -- xác thực và cấp JWT},
  label={lst:auth}]
public async Task<AuthResponseDto> LoginAsync(LoginDto dto)
{
    var account = await _context.Accounts
        .FirstOrDefaultAsync(x => x.Email == dto.Email);

    if (account == null)
        throw new Exception("Invalid email");

    // Xác minh mật khẩu bằng BCrypt
    var valid = BCrypt.Net.BCrypt.Verify(
        dto.Password, account.PasswordHash);
    if (!valid)
        throw new Exception("Invalid password");

    // Lấy danh sách Role từ bảng AccountRoles
    var roles = await _context.AccountRoles
        .Where(x => x.AccountId == account.AccountId)
        .Select(x => x.Role.RoleName)
        .ToListAsync();

    // Tạo JWT token chứa thông tin Role
    var token = _jwtHelper.GenerateToken(account, roles);

    return new AuthResponseDto
    {
        Token = token,
        Email = account.Email,
        Role  = roles.FirstOrDefault()!
    };
}
\end{lstlisting}

\section{Lớp Nghiệp vụ — Business Logic Layer}\label{Sec3.3}
% ═════════════════════════════════════════════════════════════

Lớp nghiệp vụ (BLL) là nơi tập trung toàn bộ logic kinh doanh.
BLL không biết dữ liệu đến từ SQL Server hay bất kỳ nguồn nào
khác — chỉ giao tiếp với DAL qua Interface.

% ──────────────────────────────────────
\subsection{Interfaces và Services}\label{sub3.3.1}

\begin{de}\label{def3.4}
Interface trong BLL đóng vai trò \textit{contract}: định nghĩa
những gì một Service phải cung cấp mà không quan tâm đến cách
thực hiện. Nhờ đó, Controller (API) chỉ phụ thuộc vào Interface,
không phụ thuộc vào implementation cụ thể — tuân thủ nguyên tắc
Dependency Inversion (chữ D trong SOLID).
\end{de}

Đoạn~\ref{lst:interface} trình bày \texttt{ITourService} với
đầy đủ các phương thức, bao gồm cả tìm kiếm qua Stored
Procedure và lấy tour phổ biến từ View.

\begin{lstlisting}[style=csharp,
  caption={ITourService -- Interface của BLL},
  label={lst:interface}]
public interface ITourService
{
    // Lấy danh sách có phân trang + lọc (LINQ to Entities)
    Task<PagedResult<TourListDto>> GetToursAsync(

        int? cateId, int? desId,
        decimal? priceMin, decimal? priceMax);

    // Chi tiết tour (Include Category, Destination, Schedule)

        string? destination, decimal? priceMin,

    // Tour phổ biến từ vw_PopularTours (EF SqlQueryRaw)
    Task<IEnumerable<PopularTourResult>> GetPopularToursAsync();

    // CRUD (chỉ Admin)

    Task<TourDetailDto> UpdateTourAsync(
        int tourId, TourRequestDto dto);
    Task DeleteTourAsync(int tourId);  // Soft-delete
}
\end{lstlisting}

\subsection{Business Rules}\label{sub3.3.2}

\begin{de}\label{def3.5}
Business Rules là các ràng buộc nghiệp vụ phản ánh quy tắc
hoạt động thực tế của doanh nghiệp. Trong kiến trúc 3 tầng,
business rules được kiểm tra tại BLL trước khi gọi xuống DAL,
đảm bảo tính hợp lệ về nghiệp vụ — không chỉ về cấu trúc dữ liệu.
\end{de}

Bảng~\ref{table:bizrules} tổng hợp các business rules

\begin{table}[H]
  \begin{center}
    \begin{tabular}{|c|l|l|}

      1 & \texttt{CateId} phải tồn tại khi tạo/sửa tour

      2 & \texttt{DesId} phải tồn tại khi tạo/sửa tour

Đoạn~\ref{lst:bizrule} minh họa các business rules trong
\texttt{TourService.CreateTourAsync}: kiểm tra tồn tại của
\texttt{CateId} và \texttt{DesId} trước khi lưu vào database.

\begin{lstlisting}[style=csharp,
  caption={Business Rules trong TourService.CreateTourAsync},
  label={lst:bizrule}]
public async Task<TourDetailDto> CreateTourAsync(TourRequestDto dto)
{
    // Business Rule 1: CateId phải tồn tại
    if (!await _cateRepo.ExistsAsync(dto.CateId))
        throw new KeyNotFoundException(
            $"Danh muc ID {dto.CateId} khong ton tai.");

    // Business Rule 2: DesId phải tồn tại
    if (!await _desRepo.ExistsAsync(dto.DesId))
        throw new KeyNotFoundException(
            $"diem den ID {dto.DesId} khong ton tai.");

    var tour = _mapper.Map<Tour>(dto);
    tour.IsActive = true;   // Mac dinh active khi tao moi

    var created = await _tourRepo.AddAsync(tour);
    var detail  = await _tourRepo.GetDetailAsync(created.TourId);
    return _mapper.Map<TourDetailDto>(detail!);
}

public async Task DeleteTourAsync(int tourId)
{
    var tour = await _tourRepo.GetByIdAsync(tourId)
        ?? throw new KeyNotFoundException(
            $"Tour ID {tourId} khong ton tai.");

    // Business Rule 3: Soft-delete thay vì xóa vật lý
    // Dữ liệu lịch sử booking vẫn được bảo toàn
    tour.IsActive = false;
    await _tourRepo.UpdateAsync(tour);
}
\end{lstlisting}

\subsection{LINQ to Objects}\label{sub3.3.3}

\begin{de}\label{def3.6}
LINQ to Objects là kỹ thuật dùng cú pháp LINQ để truy vấn
các collection \texttt{IEnumerable<T>} đang tồn tại trong
bộ nhớ (RAM), không phát sinh truy vấn SQL xuống database.
BLL dùng LINQ to Objects để tính toán, lọc và biến đổi
dữ liệu đã được DAL trả về.
\end{de}

Đoạn~\ref{lst:linqobj} trích từ \texttt{TourService.GetToursAsync}:
sau khi DAL trả về danh sách \texttt{Tour}, BLL dùng
LINQ to Objects để map sang DTO và tính \texttt{AvgRating}
từ collection \texttt{Reviews} đã được Include.

    : AbstractValidator<CreateBookingRequestDto>

    public BookingValidator()
    {

            .GreaterThan(0)
            .WithMessage("AccountId la bat buoc.");

        RuleFor(x => x.ScheduleId)
            .GreaterThan(0)
            .WithMessage("ScheduleId la bat buoc.");

        RuleFor(x => x.NumberOfPeople)
            .GreaterThan(0)
            .WithMessage("NumberOfPeople phai lon hon 0.");

        // Kiểm tra số hành khách khớp với số người đặt
        RuleFor(x => x.Passengers)

                list != null && list.Count == dto.NumberOfPeople)

                "So hanh khach phai bang NumberOfPeople.");

        // It nhat 1 khach hang phai la lien lac chinh

            .Must(x => x.Passengers.Any(p => p.IsPrimaryContact))
            .WithMessage("Phai co it nhat 1 lien lac chinh.");
    }
}

\subsection{LINQ to Entities và Repository Pattern}
  \label{sub3.4.2}

\begin{thm}\label{thm3.3}
LINQ to Entities dịch biểu thức lambda C\# thành câu SQL tối
ưu tại runtime thông qua Expression Tree. Cơ chế này đảm bảo
type-safe và tự động parameterize tham số, hoàn toàn ngăn chặn
SQL Injection mà không cần lập trình viên xử lý thủ công.
\end{thm}

\noindent\begin{proof}
Biểu thức \texttt{Where(t => t.Price >= priceMin)} không thực
thi ngay mà tạo một Expression Tree. Đến khi gọi
\texttt{ToListAsync()}, EF Core dịch thành
\texttt{WHERE Price >= @p0} với \texttt{@p0} là
\texttt{SqlParameter} — không cho phép nhúng giá trị trực tiếp
vào câu SQL, triệt tiêu nguy cơ SQL Injection. \QEDAa
\end{proof}

\texttt{GenericRepository<T>} (Đoạn~\ref{lst:genrepo}) cung
cấp các thao tác CRUD cơ bản cho mọi Entity.
\texttt{TourRepository} kế thừa và bổ sung các query đặc thù.

\begin{lstlisting}[style=csharp,
  caption={GenericRepository<T> -- CRUD cơ bản với EF Core},
  label={lst:genrepo}]
public class GenericRepository<T> : IRepository<T>
    where T : class
{
    protected readonly AppDbContext _db;

    public GenericRepository(AppDbContext db) { _db = db; }

    public async Task<IEnumerable<T>> GetAllAsync() =>
        await _db.Set<T>().ToListAsync();

    public async Task<T?> GetByIdAsync(int id) =>
        await _db.Set<T>().FindAsync(id);

    public async Task<T> AddAsync(T entity)
    {
        _db.Set<T>().Add(entity);
        await _db.SaveChangesAsync();
        return entity;
    }

    public async Task UpdateAsync(T entity)
    {
        _db.Set<T>().Update(entity);
        await _db.SaveChangesAsync();
    }

    public async Task DeleteAsync(int id)
    {
        var entity = await GetByIdAsync(id);
        if (entity is not null)
        {
            _db.Set<T>().Remove(entity);
            await _db.SaveChangesAsync();
        }
    }

    public async Task<bool> ExistsAsync(int id) =>
        await _db.Set<T>().FindAsync(id) is not null;
}
\end{lstlisting}

minh họa LINQ to Entities với filter động nhiều điều kiện,
Include navigation properties và phân trang.

\begin{lstlisting}[style=csharp,
  caption={LINQ to Entities trong TourRepository.GetPagedAsync},
  label={lst:linq}]
public async Task<(IEnumerable<Tour> Items, int TotalCount)>
    GetPagedAsync(
        int page, int pageSize,
        int? cateId, int? desId,
        decimal? priceMin, decimal? priceMax)
{
    // IQueryable<T> -- chưa thực thi, chỉ build biểu thức
    var query = _db.Tours
        .Include(t => t.Category)     // JOIN Categories
        .Include(t => t.Destination)  // JOIN Destinations
        .Include(t => t.Reviews)      // JOIN Reviews
        .Where(t => t.IsActive);

    // Filter động -- chỉ thêm WHERE khi tham số không null
    if (cateId.HasValue)
        query = query.Where(t => t.CateId == cateId.Value);
    if (desId.HasValue)
        query = query.Where(t => t.DesId == desId.Value);
    if (priceMin.HasValue)
        query = query.Where(t => t.Price >= priceMin.Value);
    if (priceMax.HasValue)
        query = query.Where(t => t.Price <= priceMax.Value);

    // COUNT trước khi phân trang (không thực thi SELECT)
    var total = await query.CountAsync();

    // OFFSET/FETCH qua Skip/Take -- thực thi tại đây
    var items = await query
        .OrderByDescending(t => t.TourId)
        .Skip((page - 1) * pageSize)
        .Take(pageSize)
        .ToListAsync();

    return (items, total);
}
\end{lstlisting}

\subsection{ADO.NET — Connected và Disconnected Model}
  \label{sub3.4.4}

\begin{de}\label{def3.8}
ADO.NET cung cấp hai mô hình truy cập dữ liệu. \textbf{Connected
model} duy trì kết nối mở trong suốt quá trình đọc dữ liệu qua
\texttt{SqlDataReader} — phù hợp đọc luồng dữ liệu lớn.
\textbf{Disconnected model} tải toàn bộ dữ liệu vào
\texttt{DataSet} qua \texttt{SqlDataAdapter} rồi ngắt kết nối

\begin{lstlisting}[style=csharp,
  caption={ADO.NET Connected Model -- gọi sp\_SearchTours},
  label={lst:adocon}]
// Connected model: kết nối mở trong khi SqlDataReader đọc
public async Task<IEnumerable<SearchTourResult>> SearchAsync(
    string? destination,
    decimal? priceMin, decimal? priceMax,
    DateOnly? date)
{
    var results = new List<SearchTourResult>();

    await using var conn = new SqlConnection(_connectionString);
    await using var cmd = new SqlCommand("sp_SearchTours", conn)
    {
        CommandType = CommandType.StoredProcedure
    };

    // Truyền tham số -- DBNull.Value khi null (optional filter)
    cmd.Parameters.AddWithValue("@Destination",
        (object?)destination ?? DBNull.Value);
    cmd.Parameters.AddWithValue("@PriceMin",
        (object?)priceMin ?? DBNull.Value);
    cmd.Parameters.AddWithValue("@PriceMax",
        (object?)priceMax ?? DBNull.Value);
    cmd.Parameters.AddWithValue("@Date",
        date.HasValue
            ? (object)date.Value.ToDateTime(TimeOnly.MinValue)
            : DBNull.Value);

    await conn.OpenAsync();
    await using var reader = await cmd.ExecuteReaderAsync();

    // Đọc từng dòng theo luồng
    while (await reader.ReadAsync())
    {
        results.Add(new SearchTourResult
        {
            TourId    = reader.GetInt32("TourId"),
            TourName  = reader.IsDBNull("TourName")
                ? null : reader.GetString("TourName"),
            Price     = reader.GetDecimal("Price"),
            DesName   = reader.IsDBNull("DesName")
                ? null : reader.GetString("DesName"),
            ScheduleId = reader.GetInt32("ScheduleId"),
            DepartureDate = DateOnly.FromDateTime(
                reader.GetDateTime("DepartureDate")),
            AvailableSlots = reader.GetInt32("AvailableSlots")
        });
    }
    return results;
}

public DataSet GetToursDataSet()
{
    var ds = new DataSet("TourCatalog");

    using var conn = new SqlConnection(_connectionString);

    // Adapter 1: nạp bảng Tours
    using var adapter = new SqlDataAdapter(
        @"SELECT T.TourId, T.TourName, T.Price,
                 C.CateName, D.DesName
          FROM   Tours T
          LEFT JOIN Categories   C ON T.CateId = C.CateId
          LEFT JOIN Destinations D ON T.DesId  = D.DesId
          WHERE  T.IsActive = 1",
        conn);
    adapter.Fill(ds, "Tours");

    // Adapter 2: nạp bảng Schedules vào cùng DataSet
    using var schedAdapter = new SqlDataAdapter(
        @"SELECT ScheduleId, TourId, DepartureDate,
                 ReturnDate, AvailableSlots, Status
          FROM   TourSchedules WHERE Status = N'Open'",
        conn);
    schedAdapter.Fill(ds, "Schedules");

    ds.Relations.Add("Tour_Schedules",
        ds.Tables["Tours"]!.Columns["TourId"]!,
        ds.Tables["Schedules"]!.Columns["TourId"]!);

    return ds;  // Trả về DataSet độc lập với DB
}

\subsection{Data Transfer Objects (DTO)}\label{sub3.5.1}

\begin{thm}\label{thm3.4}
Việc sử dụng DTO thay vì trả Entity trực tiếp ra API là bắt
buộc trong kiến trúc nhiều tầng vì ba lý do: (1) Entity chứa
thông tin nhạy cảm như \texttt{PasswordHash}; (2) Navigation
properties gây vòng lặp tuần hoàn khi JSON serialize; (3) Cần
kiểm soát chính xác dữ liệu nào được phép hiển thị theo Role.
\end{thm}

\noindent\begin{proof}
Entity \texttt{Account} chứa \texttt{PasswordHash} -- nếu
serialize trực tiếp, client nhận được chuỗi BCrypt hash.
\texttt{AuthResponseDto} chỉ chứa \texttt{Token}, \texttt{Email},
\texttt{Role} -- thông tin nhạy cảm được che giấu hoàn toàn.
\QEDAa
\end{proof}

\subsection{AutoMapper}\label{sub3.5.2}

\begin{de}\label{def3.9}
AutoMapper là thư viện tự động ánh xạ (map) giữa các object
có cấu trúc tương tự, loại bỏ code mapping thủ công lặp lại.

\texttt{ForMember} cho các trường hợp cần tùy chỉnh.

Đoạn~\ref{lst:mapper} trình bày \texttt{MappingProfile} thực

\texttt{IsActive} khi map \texttt{TourRequestDto} sang
\texttt{Tour}.

\begin{lstlisting}[style=csharp,
  caption={MappingProfile -- AutoMapper thực tế của hệ thống},
  label={lst:mapper}]
public class MappingProfile : Profile
{
    public MappingProfile()
    {
        // Category: map cả hai chiều
        CreateMap<Category, CategoryResponseDto>();
        CreateMap<CategoryRequestDto, Category>()
            .ForMember(d => d.CateId, o => o.Ignore());

        // TourSchedule -> ScheduleResponseDto
        // Lấy tên nhân viên từ navigation property
        CreateMap<TourSchedule, ScheduleResponseDto>()
            .ForMember(d => d.EmployeeName, o => o.MapFrom(s =>
                s.Employee != null ? s.Employee.FullName : null));

        // Tour -> TourListDto: tính AvgRating từ Reviews
        CreateMap<Tour, TourListDto>()
            .ForMember(d => d.CateName, o => o.MapFrom(t =>
                t.Category != null ? t.Category.CateName : null))
            .ForMember(d => d.DesName, o => o.MapFrom(t =>
                t.Destination != null
                    ? t.Destination.DesName : null))
            .ForMember(d => d.AvgRating, o => o.MapFrom(t =>
                t.Reviews.Any()
                    ? Math.Round(
                        t.Reviews.Average(r => (double)r.Rating),1)
                    : (double?)null));

        // Tour -> TourDetailDto: thêm Country, City, Schedules
        CreateMap<Tour, TourDetailDto>()
            .ForMember(d => d.Country, o => o.MapFrom(t =>
                t.Destination != null ? t.Destination.Country : null))
            .ForMember(d => d.City, o => o.MapFrom(t =>
                t.Destination != null ? t.Destination.City : null))
            .ForMember(d => d.TotalReviews,
                o => o.MapFrom(t => t.Reviews.Count))
            .ForMember(d => d.Schedules,
                o => o.MapFrom(t => t.TourSchedules));

        // TourRequestDto -> Tour: ignore TourId và IsActive
        // (IsActive được set thủ công trong Service)
        CreateMap<TourRequestDto, Tour>()
            .ForMember(d => d.TourId,   o => o.Ignore())
            .ForMember(d => d.IsActive, o => o.Ignore());
    }
}
\end{lstlisting}

\subsection{Chuẩn hóa Response với ApiResponse<T>}\label{sub3.5.3}

\begin{de}\label{def3.10}
\texttt{ApiResponse<T>} là wrapper chuẩn hóa toàn bộ response
của API. Mọi endpoint đều trả về cùng một cấu trúc JSON, giúp
client xử lý response nhất quán và dễ debug khi có lỗi.
\end{de}

\begin{lstlisting}[style=csharp,
  caption={ApiResponse<T> -- wrapper chuẩn hóa response},
  label={lst:apiresponse}]
public class ApiResponse<T>
{
    public bool    Success { get; init; }
    public T?      Data    { get; init; }
    public string? Message { get; init; }

    // Tạo response thành công
    public static ApiResponse<T> Ok(T? data,
        string? message = null) =>
        new ApiResponse<T>
        {
            Success = true, Data = data, Message = message
        };

    // Tạo response lỗi
    public static ApiResponse<T> Fail(
        string? message = null) =>
        new ApiResponse<T>
        {
            Success = false, Data = default, Message = message
        };
}
\end{lstlisting}

\section{Dependency Injection và cấu hình hệ thống}\label{Sec3.6}
% ═════════════════════════════════════════════════════════════

\begin{de}\label{def3.11}
Dependency Injection (DI) là cơ chế kết nối các lớp trong
kiến trúc 3 tầng mà không tạo sự phụ thuộc trực tiếp.
Controller không tự \texttt{new TourService()} mà nhận
\texttt{ITourService} qua constructor -- ASP.NET Core DI
Container tự tạo và inject instance phù hợp theo vòng đời
đã đăng ký.
\end{de}

\begin{lem}\label{lem3.1}
Repositories và Services đăng ký với vòng đời
\texttt{Scoped}: mỗi HTTP request nhận một instance riêng,
nhưng mọi class trong cùng request đều chia sẻ chung một
\texttt{AppDbContext}. Điều này đảm bảo Unit of Work:
tất cả thay đổi trong một request được commit hoặc rollback
cùng nhau thông qua một \texttt{SaveChangesAsync()}.
\end{lem}

\begin{lstlisting}[style=csharp,
  caption={Program.cs -- đăng ký DI đầy đủ của hệ thống},
  label={lst:di}]
var builder = WebApplication.CreateBuilder(args);
var cfg = builder.Configuration;

// ── 1. Database (Scoped DbContext)
builder.Services.AddDbContext<AppDbContext>(opt =>
    opt.UseSqlServer(
        cfg.GetConnectionString("DefaultConnection")));

// ── 2. ADO.NET (Singleton -- không dùng DbContext)
builder.Services.AddSingleton(_ =>
    new AdoTourRepository(
        cfg.GetConnectionString("DefaultConnection")!));

// ── 3. DAL Repositories (Scoped)
builder.Services.AddScoped<ITourRepository,        TourRepository>();
builder.Services.AddScoped<ICategoryRepository,    CategoryRepository>();
builder.Services.AddScoped<IDestinationRepository, DestinationRepository>();
builder.Services.AddScoped<IScheduleRepository,    ScheduleRepository>();
builder.Services.AddScoped<IBookingRepository,     BookingRepository>();
builder.Services.AddScoped<IPaymentRepository,     PaymentRepository>();
builder.Services.AddScoped<IReportRepository,      ReportRepository>();

// ── 4. BLL Services (Scoped)
builder.Services.AddScoped<ITourService,       TourService>();
builder.Services.AddScoped<ICategoryService,   CategoryService>();
builder.Services.AddScoped<IBookingService,    BookingService>();
builder.Services.AddScoped<IAuthService,       AuthService>();
builder.Services.AddScoped<IReviewService,     ReviewService>();
builder.Services.AddScoped<IReportService,     ReportService>();
builder.Services.AddScoped<IPaymentService,    PaymentService>();
builder.Services.AddScoped<JwtHelper>();

// ── 5. AutoMapper
builder.Services.AddAutoMapper(typeof(MappingProfile));

// ── 6. FluentValidation
builder.Services.AddFluentValidationAutoValidation();
builder.Services.AddValidatorsFromAssemblyContaining
    <BookingValidator>();

// ── 7. JWT Authentication
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        var key = Encoding.UTF8.GetBytes(
            cfg["JwtSettings:SecretKey"]!);
        options.TokenValidationParameters =
            new TokenValidationParameters
            {
                ValidateIssuer           = true,
                ValidateAudience         = true,
                ValidateLifetime         = true,
                ValidateIssuerSigningKey = true,
                ValidIssuer    = cfg["JwtSettings:Issuer"],
                ValidAudience  = cfg["JwtSettings:Audience"],
                IssuerSigningKey =
                    new SymmetricSecurityKey(key)
            };
    });
builder.Services.AddAuthorization();

// ── 8. CORS cho Angular
builder.Services.AddCors(opt =>
    opt.AddPolicy("AllowAngular", p =>
        p.WithOrigins("http://localhost:4200")
         .AllowAnyHeader().AllowAnyMethod()));

// ── 9. Swagger với JWT support
builder.Services.AddSwaggerGen(s =>
{
    s.SwaggerDoc("v1", new OpenApiInfo {
        Title   = "TravelTourBooking API",
        Version = "v1"
    });
    s.AddSecurityDefinition("Bearer",
        new OpenApiSecurityScheme {
            Name        = "Authorization",
            Type        = SecuritySchemeType.Http,
            Scheme      = "bearer",
            BearerFormat = "JWT",
            In          = ParameterLocation.Header
        });
});

builder.Services.AddControllers();
var app = builder.Build();

// ── Middleware pipeline (thứ tự bắt buộc)
app.UseMiddleware<ExceptionMiddleware>(); // Đầu tiên
app.UseSwagger();
app.UseSwaggerUI();
app.UseHttpsRedirection();
app.UseCors("AllowAngular");
app.UseAuthentication();  // Trước Authorization
app.UseAuthorization();
app.MapControllers();
app.Run();
\end{lstlisting}

\begin{rem}
\texttt{AdoTourRepository} được đăng ký với vòng đời
\texttt{Singleton} (không phải \texttt{Scoped}) vì nó không
dùng \texttt{AppDbContext} — mỗi method tự quản lý
\texttt{SqlConnection} riêng thông qua \texttt{await using},
đảm bảo connection được đóng ngay khi ra khỏi phạm vi sử dụng.
\end{rem}

\section{LINQ to XML -- Xuất và Nhập dữ liệu}\label{Sec3.7}
% ═════════════════════════════════════════════════════════════

\begin{de}\label{def3.12}
LINQ to XML là API trong .NET cho phép tạo, truy vấn và biến
đổi tài liệu XML bằng cú pháp LINQ quen thuộc thông qua các
lớp \texttt{XDocument} và \texttt{XElement}. So với
\texttt{XmlDocument} truyền thống, LINQ to XML có cú pháp
ngắn gọn hơn, hỗ trợ functional construction (tạo cây XML
bằng một biểu thức duy nhất) và tích hợp trực tiếp với
LINQ to Objects.
\end{de}

Hệ thống sử dụng \texttt{XmlSerializer} trong
\texttt{XmlService} (Đoạn~\ref{lst:xmlservice}) để
serialize và deserialize object C\# sang định dạng XML
chuẩn, phục vụ chức năng xuất nhập dữ liệu tour.

\begin{lstlisting}[style=csharp,
  caption={XmlService -- Export và Import dữ liệu XML},
  label={lst:xmlservice}]
public static class XmlService
{
    // Serialize object sang chuỗi XML (Export)
    public static string ToXml<T>(T obj)
    {
        var xs = new XmlSerializer(typeof(T));
        using var ms = new MemoryStream();
        xs.Serialize(ms, obj);
        return Encoding.UTF8.GetString(ms.ToArray());
    }
}
\end{lstlisting}

\begin{ex}\label{ex3.1}
Để xuất danh sách tour ra XML, Controller gọi
\texttt{XmlService.ToXml(tours)} -- \texttt{XmlSerializer}
tự động ánh xạ từng property của \texttt{TourListDto} thành
phần tử XML tương ứng. File XML kết quả có thể được lưu
trữ, trao đổi với hệ thống khác hoặc nhập lại vào hệ thống.
\end{ex}
